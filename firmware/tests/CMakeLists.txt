cmake_minimum_required(VERSION 3.16)
project(counter_host_tests LANGUAGES CXX)
enable_testing()
add_executable(counter_tests test_counter.cpp)
target_compile_features(counter_tests PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(counter_tests PRIVATE /W4 /UNDEBUG)
else()
  target_compile_options(counter_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
endif()
add_test(NAME counter_behavior COMMAND counter_tests)

add_executable(shift_tests test_shift.cpp)
target_compile_features(shift_tests PRIVATE cxx_std_17)
if(MSVC)
  target_compile_options(shift_tests PRIVATE /W4 /UNDEBUG)
else()
  target_compile_options(shift_tests PRIVATE -Wall -Wextra -Werror -UNDEBUG)
endif()
add_test(NAME shift_register_hardware_model COMMAND shift_tests)
